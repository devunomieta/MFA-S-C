import { useEffect, useState } from "react";

import { MessageSquare, Plus, Trash2, Layout, CheckCircle2, XCircle } from "lucide-react";
import { toast } from "sonner";

import { Badge } from "@/app/components/ui/badge";
import { Button } from "@/app/components/ui/button";
import {
  Card,
  CardContent,
  CardHeader,
  CardTitle,
  CardDescription,
} from "@/app/components/ui/card";
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
  DialogDescription,
  DialogFooter,
} from "@/app/components/ui/dialog";
import { Input } from "@/app/components/ui/input";
import { Label } from "@/app/components/ui/label";
import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/app/components/ui/tabs";
import { supabase } from "@/lib/supabase";

export default function SurveyManagement() {
  const [surveys, setSurveys] = useState<any[]>([]);
  const [responses, setResponses] = useState<any[]>([]);
  const [loading, setLoading] = useState(true);

  const [activeTab, setActiveTab] = useState("questions");

  // New Survey Form State
  const [isNewDialogOpen, setIsNewDialogOpen] = useState(false);
  const [newSurvey, setNewSurvey] = useState({
    question: "",
    trigger_event: "first_deposit",
    options: ["", "", ""],
  });

  useEffect(() => {
    fetchData();
  }, []);

  async function fetchData() {
    setLoading(true);
    try {
      const { data: sData } = await supabase
        .from("surveys")
        .select("*")
        .order("created_at", { ascending: false });
      const { data: rData } = await supabase
        .from("survey_responses")
        .select(
          `
          *,
          profiles(full_name, email),
          surveys(question)
        `,
        )
        .order("created_at", { ascending: false });

      setSurveys(sData || []);
      setResponses(rData || []);
    } catch {
      toast.error("Failed to load survey data");
    } finally {
      setLoading(false);
    }
  }

  async function handleCreateSurvey() {
    if (!newSurvey.question || newSurvey.options.some((o) => !o)) {
      toast.error("Please fill all fields and options");
      return;
    }

    try {
      const { error } = await supabase.from("surveys").insert({
        question: newSurvey.question,
        trigger_event: newSurvey.trigger_event,
        options: newSurvey.options.filter((o) => !!o),
      });

      if (error) throw error;
      toast.success("Survey created!");
      setIsNewDialogOpen(false);
      fetchData();
    } catch (err: any) {
      toast.error(err.message);
    }
  }

  async function toggleSurvey(id: string, current: boolean) {
    const { error } = await supabase.from("surveys").update({ is_active: !current }).eq("id", id);
    if (error) toast.error("Update failed");
    else fetchData();
  }

  async function deleteSurvey(id: string) {
    const { error } = await supabase.from("surveys").delete().eq("id", id);
    if (error) toast.error("Delete failed");
    else fetchData();
  }

  if (loading) {
    return (
      <div className="h-96 flex items-center justify-center">
        <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-emerald-600"></div>
      </div>
    );
  }

  return (
    <div className="space-y-6">
      <div className="flex flex-col md:flex-row md:items-center justify-between gap-4">
        <div>
          <h1 className="text-3xl font-bold dark:text-white">Survey Management</h1>
          <p className="text-slate-500 dark:text-slate-400">
            Manage user feedback and platform experience surveys.
          </p>
        </div>

        <Dialog open={isNewDialogOpen} onOpenChange={setIsNewDialogOpen}>
          <DialogTrigger asChild>
            <Button className="bg-emerald-600 hover:bg-emerald-700 h-11 px-6 rounded-xl font-bold">
              <Plus className="mr-2 size-5" /> New Survey
            </Button>
          </DialogTrigger>
          <DialogContent className="sm:max-w-[425px] dark:bg-slate-900 border-slate-800">
            <DialogHeader>
              <DialogTitle className="dark:text-white">Create New Survey</DialogTitle>
              <DialogDescription>Define a question and options to show users.</DialogDescription>
            </DialogHeader>
            <div className="space-y-4 py-4">
              <div className="space-y-2">
                <Label className="dark:text-white">Question</Label>
                <Input
                  placeholder="e.g. How was your experience?"
                  value={newSurvey.question}
                  onChange={(e) => setNewSurvey({ ...newSurvey, question: e.target.value })}
                  className="dark:bg-slate-800 dark:border-slate-700"
                />
              </div>
              <div className="space-y-2">
                <Label className="dark:text-white">Trigger Event</Label>
                <select
                  className="w-full h-10 rounded-md border border-slate-700 bg-slate-800 text-sm px-3 text-white"
                  value={newSurvey.trigger_event}
                  onChange={(e) => setNewSurvey({ ...newSurvey, trigger_event: e.target.value })}
                >
                  <option value="first_deposit">First Deposit</option>
                  <option value="first_plan">First Plan Join</option>
                  <option value="first_loan">First Loan Application</option>
                  <option value="first_repayment">First Loan Repayment</option>
                  <option value="first_withdrawal">First Withdrawal</option>
                </select>
              </div>
              <div className="space-y-2">
                <Label className="dark:text-white">Options</Label>
                {newSurvey.options.map((opt, i) => (
                  <div key={i} className="flex gap-2">
                    <Input
                      placeholder={`Option ${i + 1}`}
                      value={opt}
                      onChange={(e) => {
                        const next = [...newSurvey.options];
                        next[i] = e.target.value;
                        setNewSurvey({ ...newSurvey, options: next });
                      }}
                      className="dark:bg-slate-800 dark:border-slate-700"
                    />
                    {newSurvey.options.length > 2 && (
                      <Button
                        variant="ghost"
                        size="icon"
                        onClick={() => {
                          const next = newSurvey.options.filter((_, idx) => idx !== i);
                          setNewSurvey({ ...newSurvey, options: next });
                        }}
                      >
                        <XCircle className="size-4 text-red-500" />
                      </Button>
                    )}
                  </div>
                ))}
                <Button
                  variant="outline"
                  size="sm"
                  className="w-full border-dashed"
                  onClick={() =>
                    setNewSurvey({ ...newSurvey, options: [...newSurvey.options, ""] })
                  }
                >
                  <Plus className="size-4 mr-2" /> Add Option
                </Button>
              </div>
            </div>
            <DialogFooter>
              <Button
                onClick={handleCreateSurvey}
                className="bg-emerald-600 hover:bg-emerald-700 w-full font-bold"
              >
                Create Survey
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      </div>

      <Tabs value={activeTab} onValueChange={setActiveTab} className="w-full">
        <TabsList className="bg-slate-100 dark:bg-slate-800 p-1 rounded-xl w-full max-w-md">
          <TabsTrigger value="questions" className="flex-1 rounded-lg gap-2">
            <Layout className="size-4" /> Questions
          </TabsTrigger>
          <TabsTrigger value="responses" className="flex-1 rounded-lg gap-2">
            <MessageSquare className="size-4" /> Responses
          </TabsTrigger>
        </TabsList>

        <TabsContent value="questions" className="space-y-4 mt-6">
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
            {surveys.map((s) => (
              <Card
                key={s.id}
                className="dark:bg-slate-900 border-slate-200 dark:border-slate-800 overflow-hidden"
              >
                <CardHeader className="pb-3">
                  <div className="flex justify-between items-start">
                    <Badge variant={s.is_active ? "default" : "secondary"}>
                      {s.is_active ? "Active" : "Inactive"}
                    </Badge>
                    <div className="flex gap-1">
                      <Button
                        variant="ghost"
                        size="icon"
                        onClick={() => toggleSurvey(s.id, s.is_active)}
                      >
                        {s.is_active ? (
                          <XCircle className="size-4 text-amber-500" />
                        ) : (
                          <CheckCircle2 className="size-4 text-emerald-500" />
                        )}
                      </Button>
                      <Button variant="ghost" size="icon" onClick={() => deleteSurvey(s.id)}>
                        <Trash2 className="size-4 text-red-500" />
                      </Button>
                    </div>
                  </div>
                  <CardTitle className="text-lg font-bold dark:text-white mt-2">
                    {s.question}
                  </CardTitle>
                  <CardDescription className="uppercase text-[10px] tracking-wider font-bold">
                    Trigger: {s.trigger_event.replace("_", " ")}
                  </CardDescription>
                </CardHeader>
                <CardContent>
                  <div className="space-y-2">
                    {s.options.map((opt: string, i: number) => (
                      <div
                        key={i}
                        className="text-sm p-2 bg-slate-50 dark:bg-slate-800/50 rounded-lg text-slate-600 dark:text-slate-300 border border-slate-100 dark:border-slate-800"
                      >
                        {opt}
                      </div>
                    ))}
                  </div>
                </CardContent>
              </Card>
            ))}
          </div>
        </TabsContent>

        <TabsContent value="responses" className="mt-6">
          <Card className="dark:bg-slate-900 border-slate-200 dark:border-slate-800">
            <div className="overflow-x-auto">
              <table className="w-full text-left">
                <thead className="bg-slate-50 dark:bg-slate-800/50 border-b border-slate-200 dark:border-slate-800">
                  <tr>
                    <th className="px-6 py-4 text-xs font-bold uppercase tracking-wider text-slate-500">
                      User
                    </th>
                    <th className="px-6 py-4 text-xs font-bold uppercase tracking-wider text-slate-500">
                      Survey Question
                    </th>
                    <th className="px-6 py-4 text-xs font-bold uppercase tracking-wider text-slate-500">
                      Response
                    </th>
                    <th className="px-6 py-4 text-xs font-bold uppercase tracking-wider text-slate-500">
                      Date
                    </th>
                  </tr>
                </thead>
                <tbody className="divide-y divide-slate-100 dark:divide-slate-800">
                  {responses.map((r) => (
                    <tr
                      key={r.id}
                      className="hover:bg-slate-50 dark:hover:bg-slate-800/30 transition-colors"
                    >
                      <td className="px-6 py-4">
                        <div className="flex flex-col">
                          <span className="text-sm font-bold dark:text-white">
                            {r.profiles?.full_name || "Anonymous"}
                          </span>
                          <span className="text-xs text-slate-400">{r.profiles?.email}</span>
                        </div>
                      </td>
                      <td className="px-6 py-4 text-sm dark:text-slate-300 font-medium">
                        {r.surveys?.question}
                      </td>
                      <td className="px-6 py-4">
                        <Badge className="bg-blue-50 text-blue-600 dark:bg-blue-900/20 dark:text-blue-400 border-blue-100 dark:border-blue-900/30">
                          {r.answer}
                        </Badge>
                      </td>
                      <td className="px-6 py-4 text-sm text-slate-500">
                        {new Date(r.created_at).toLocaleDateString()}
                      </td>
                    </tr>
                  ))}
                  {responses.length === 0 && (
                    <tr>
                      <td colSpan={4} className="px-6 py-10 text-center text-slate-500 italic">
                        No responses yet
                      </td>
                    </tr>
                  )}
                </tbody>
              </table>
            </div>
          </Card>
        </TabsContent>
      </Tabs>
    </div>
  );
}
